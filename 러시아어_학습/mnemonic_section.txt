function updateStats() {
  const total = VOCAB.length;
  const learnedCount = Object.values(learned).filter(Boolean).length;
  const lastScore = quizScores.length ? quizScores[quizScores.length-1] : 0;
  document.getElementById('stat-words').textContent = total;
  document.getElementById('stat-learned').textContent = learnedCount;
  document.getElementById('stat-quiz').textContent = lastScore ? lastScore + '%' : '—';
  const pct = Math.round(learnedCount / total * 100);
  document.getElementById('prog-vocab').style.width = pct + '%';
  document.getElementById('prog-grammar').style.width = '0%';
}

function updateBadges() {

  VOCAB.forEach(w => { counts[w.lesson]++; });
  document.getElementById('badge-all').textContent = counts.all;
  document.getElementById('badge-1').textContent = counts['1'];
  document.getElementById('badge-2').textContent = counts['2'];
  document.getElementById('badge-3').textContent = counts['3'];
  document.getElementById('badge-conv').textContent = counts.conv;
}

function renderWordGrid(lesson) {
  currentLesson = lesson;
  const search = (document.getElementById('vocab-search')?.value || '').toLowerCase();
  const showLearnedOnly = document.getElementById('show-learned')?.checked;

  let words = VOCAB.filter(w => {
    if (lesson !== 'all' && w.lesson !== lesson) return false;
    if (search && !w.ru.toLowerCase().includes(search) && !w.ko.includes(search)) return false;
    if (showLearnedOnly && !learned[w.ru]) return false;
    return true;
  });

  const grid = document.getElementById('word-grid');
  grid.innerHTML = '';
  words.forEach(w => {
    const cell = document.createElement('div');
    cell.className = 'word-cell' + (learned[w.ru] ? ' learned' : '');
    cell.innerHTML = `<div class="word-ru">${w.ru}</div><div class="word-ko">${w.ko}</div>`;
    cell.onclick = () => toggleLearned(w.ru, cell);
    grid.appendChild(cell);
  });

  if (words.length === 0) {
    grid.innerHTML = '<div style="padding:40px;text-align:center;color:var(--muted);grid-column:1/-1;">검색 결과가 없습니다</div>';
  }
}

function toggleLearned(ru, cell) {
  learned[ru] = !learned[ru];
  localStorage.setItem('ru_learned', JSON.stringify(learned));
  cell.classList.toggle('learned', learned[ru]);
  updateStats();
  notify(learned[ru] ? '✓ 학습 완료로 표시했습니다' : '학습 완료 표시를 해제했습니다', learned[ru] ? 'success' : '');
}

function filterVocab(lesson, btn) {
  currentLesson = lesson;
  if (btn) {
    document.querySelectorAll('#vocab-tabs .lesson-tab').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
  }
  renderWordGrid(lesson);
}

function filterBySearch() { renderWordGrid(currentLesson); }

function resetLearned() {
  if (!confirm('모든 학습 완료 기록을 초기화할까요?')) return;
  learned = {};
  localStorage.removeItem('ru_learned');
  renderWordGrid(currentLesson);
  updateStats();
  notify('초기화되었습니다', 'warning');
}

function getFlashDeck(lesson) {
  if (lesson === 'all') return [...VOCAB];
  return VOCAB.filter(w => w.lesson === lesson);
}

function renderFlashCards(lesson) {
  currentFlashLesson = lesson;
  flashCards = [...getFlashDeck(lesson)];
  flashIdx = 0;
  flashKnown.clear();
  flashUnknown.clear();
  showFlashCard();
}

function setFlashLesson(lesson, btn) {
  document.querySelectorAll('#sec-flash .lesson-tab').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
  renderFlashCards(lesson);
}

function setFlashDir() {
  flashDir = document.getElementById('flash-dir').checked;
  showFlashCard();
}

function showFlashCard() {
  const card = document.getElementById('flashcard');
  card.classList.remove('flipped');

  const total = flashCards.length;
  document.getElementById('flash-info').textContent = `카드 ${Math.min(flashIdx+1, total)} / ${total}`;

  if (total === 0) {
    document.getElementById('flash-front').textContent = '단어 없음';
    document.getElementById('flash-back').textContent = '—';
    document.getElementById('flash-lesson-tag').textContent = '';
    document.getElementById('flash-hint').textContent = '';
    return;
  }

  const w = flashCards[flashIdx % total];
  if (!flashDir) {
    document.getElementById('flash-front').textContent = w.ru;
    document.getElementById('flash-back').textContent = w.ko;
  } else {
    document.getElementById('flash-front').textContent = w.ko;
    document.getElementById('flash-back').textContent = w.ru;
  }
  document.getElementById('flash-lesson-tag').textContent = w.lesson === 'conv' ? '회화' : w.lesson + '과';
  document.getElementById('flash-hint').textContent = '';

  const known = flashKnown.size;
  const unknown = flashUnknown.size;
  const answered = known + unknown;
  document.getElementById('flash-known').textContent = '알아요: ' + known;
  document.getElementById('flash-unknown').textContent = '모름: ' + unknown;
  document.getElementById('flash-prog').style.width = answered > 0 ? (known/answered*100) + '%' : '0%';
}

function flipCard() {
  document.getElementById('flashcard').classList.toggle('flipped');
}

function nextCard() {
  flashIdx = (flashIdx + 1) % flashCards.length;
  showFlashCard();
}

function prevCard() {
  flashIdx = (flashIdx - 1 + flashCards.length) % flashCards.length;
  showFlashCard();
}

function markCard(known) {
  if (flashCards.length === 0) return;
  const w = flashCards[flashIdx % flashCards.length];
  if (known) {
    flashKnown.add(w.ru);
    flashUnknown.delete(w.ru);
    notify('✓ 알아요!', 'success');
  } else {
    flashUnknown.add(w.ru);
    flashKnown.delete(w.ru);
    notify('다시 공부하세요');
  }
  nextCard();
}

function shuffleCards() {
  for (let i = flashCards.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [flashCards[i], flashCards[j]] = [flashCards[j], flashCards[i]];
  }
  flashIdx = 0;
  showFlashCard();
  notify('🔀 카드를 섞었습니다!');
}

function resetCards() {
  renderFlashCards(currentFlashLesson);
  notify('↺ 처음부터 시작합니다');
}

function onlyUnknown() {
  if (flashUnknown.size === 0) { notify('모르는 카드가 없어요! 🎉', 'success'); return; }
  flashCards = VOCAB.filter(w => flashUnknown.has(w.ru));
  flashIdx = 0;
  flashKnown.clear();
  flashUnknown.clear();
  showFlashCard();
  notify(`모르는 단어 ${flashCards.length}개만 표시합니다`);
}

function selectQuizOpt(btn, field, value) {
  const parent = btn.closest('.quiz-options');
  parent.querySelectorAll('.quiz-option-btn').forEach(b => b.classList.remove('selected'));
  btn.classList.add('selected');
  quizState[field] = value;
}

function startQuiz() {
  let pool = VOCAB;
  if (quizState.lesson !== 'all') pool = pool.filter(w => w.lesson === quizState.lesson);

  if (pool.length < 4) { notify('단어가 너무 적습니다 (최소 4개 필요)'); return; }

  const shuffled = [...pool].sort(() => Math.random() - 0.5);
  quizState.questions = shuffled.slice(0, Math.min(20, shuffled.length));
  quizState.currentQ = 0;
  quizState.score = 0;
  quizState.wrongAnswers = [];
  quizState.pool = pool;

  renderQuizQuestion();
}

function renderQuizQuestion() {
  const area = document.getElementById('quiz-area');
  const {questions, currentQ, mode, direction, pool} = quizState;

  if (currentQ >= questions.length) { renderQuizResult(); return; }

  const q = questions[currentQ];
  const question = direction === 'ru2ko' ? q.ru : q.ko;
  const answer = direction === 'ru2ko' ? q.ko : q.ru;
  const pct = Math.round(currentQ / questions.length * 100);

  let inputHTML = '';
  if (mode === 'mc') {
    const others = pool.filter(w => w.ru !== q.ru).sort(() => Math.random() - 0.5).slice(0, 3);
    const choices = [...others.map(w => direction === 'ru2ko' ? w.ko : w.ru), answer].sort(() => Math.random() - 0.5);
    inputHTML = `<div class="quiz-choices">` + choices.map(c =>
      `<button class="quiz-choice" onclick="checkMC(this,'${c.replace(/'/g,"\\'")}','${answer.replace(/'/g,"\\'")}')">${c}</button>`
    ).join('') + `</div>`;
  } else {
    inputHTML = `<div class="quiz-input-wrap">
      <input type="text" id="quiz-input" placeholder="정답을 입력하세요..." onkeydown="if(event.key==='Enter')checkFill()">
      <br><button class="btn btn-primary" onclick="checkFill()">확인</button>
    </div>`;
  }

  area.innerHTML = `
    <div class="quiz-card">
      <div class="quiz-q-num">문제 ${currentQ+1} / ${questions.length}</div>
      <div class="quiz-q-progress"><div class="quiz-q-progress-fill" style="width:${pct}%"></div></div>
      <div class="quiz-question">${question}</div>
      ${inputHTML}
      <div id="quiz-feedback" class="quiz-feedback" style="display:none;"></div>
      <button id="next-btn" class="btn btn-primary" style="width:100%;margin-top:16px;display:none;" onclick="nextQuestion()">다음 문제 →</button>
    </div>
  `;

  if (mode === 'fill') setTimeout(() => document.getElementById('quiz-input')?.focus(), 50);
}

function checkMC(btn, chosen, answer) {
  if (quizState.answered) return;
  quizState.answered = true;

  document.querySelectorAll('.quiz-choice').forEach(b => {
    b.classList.add('disabled');
    if (b.textContent === answer) b.classList.add('correct');
  });

  const correct = chosen === answer;
  if (!correct) {
    btn.classList.add('wrong');
    quizState.wrongAnswers.push(quizState.questions[quizState.currentQ]);
  } else {
    quizState.score++;
  }

  const fb = document.getElementById('quiz-feedback');
  fb.style.display = 'block';
  fb.className = 'quiz-feedback ' + (correct ? 'correct' : 'wrong');
  fb.textContent = correct ? '✓ 정답!' : `✗ 오답. 정답: ${answer}`;

  document.getElementById('next-btn').style.display = 'block';
  quizState.answered = false;
}

function checkFill() {
  const input = document.getElementById('quiz-input');
  if (!input) return;
  const val = input.value.trim();
  if (!val) return;

  const q = quizState.questions[quizState.currentQ];
  const answer = quizState.direction === 'ru2ko' ? q.ko : q.ru;
  const correct = val.toLowerCase() === answer.toLowerCase();

  input.className = correct ? 'correct' : 'wrong';
  input.disabled = true;

  if (correct) {
    quizState.score++;
  } else {
    quizState.wrongAnswers.push(q);
  }

  const fb = document.getElementById('quiz-feedback');
  fb.style.display = 'block';
  fb.className = 'quiz-feedback ' + (correct ? 'correct' : 'wrong');
  fb.textContent = correct ? '✓ 정답!' : `✗ 오답. 정답: ${answer}`;

  document.getElementById('next-btn').style.display = 'block';
}

function nextQuestion() {
  quizState.currentQ++;
  renderQuizQuestion();
}

function renderQuizResult() {
  const area = document.getElementById('quiz-area');
  const {score, questions, wrongAnswers} = quizState;
  const pct = Math.round(score / questions.length * 100);

  quizScores.push(pct);
  if (quizScores.length > 20) quizScores.shift();
  localStorage.setItem('ru_scores', JSON.stringify(quizScores));
  updateStats();

  let gradeClass = 'grade-A', gradeText = 'A+ 완벽해요! 🎉';
  if (pct < 90) { gradeClass = 'grade-B'; gradeText = 'B 잘하고 있어요! 💪'; }
  if (pct < 70) { gradeClass = 'grade-C'; gradeText = 'C 더 연습하세요 📚'; }
  if (pct < 50) { gradeClass = 'grade-F'; gradeText = 'F 다시 공부해요 😢'; }

  const wrongHTML = wrongAnswers.length > 0 ? `
    <div class="wrong-list">
      <div style="font-size:13px;font-weight:700;color:var(--danger);margin-bottom:10px;">틀린 단어 (${wrongAnswers.length}개)</div>
      ${wrongAnswers.map(w => `<div class="wrong-item"><span class="ru">${w.ru}</span><span class="ko">${w.ko}</span></div>`).join('')}
    </div>
  ` : '<div style="color:var(--success);font-size:14px;margin-top:16px;">🎉 모두 맞혔어요!</div>';

  area.innerHTML = `
    <div class="quiz-result">
      <div class="score">${pct}%</div>
      <div class="score-label">${score} / ${questions.length} 정답</div>
      <div class="grade ${gradeClass}">${gradeText}</div>
      ${wrongHTML}
      <div style="display:flex;gap:12px;justify-content:center;margin-top:24px;flex-wrap:wrap;">
        <button class="btn btn-primary" onclick="startQuiz()">다시 풀기</button>
        <button class="btn btn-outline" onclick="renderQuizSetup()">설정 변경</button>
        ${wrongAnswers.length > 0 ? `<button class="btn btn-danger" onclick="quizWrongOnly()">틀린 것만 다시</button>` : ''}
      </div>
    </div>
  `;
}

function quizWrongOnly() {
  if (quizState.wrongAnswers.length < 2) { notify('틀린 단어가 부족합니다 (최소 2개)'); return; }
  const pool = quizState.wrongAnswers;
  quizState.questions = [...pool].sort(() => Math.random() - 0.5);
  quizState.pool = pool;
  quizState.currentQ = 0;
  quizState.score = 0;
  quizState.wrongAnswers = [];
  renderQuizQuestion();
}
